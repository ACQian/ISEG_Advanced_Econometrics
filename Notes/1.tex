\section{Likelihood function and the ML principle}

Let $f(y;\theta_0)$ denote the probability density function of the random variable $y$ given $\theta_0$.
Our objective is to estimate the true parameter vector $\theta_0$.
The joint density of $n$ iid observations of y is
\begin{equation}
    f(y_1,\ldots,y_n|\theta_0) = \prod_{i=1}^{n}f(y_i;\theta_0)
\end{equation}
Now taking $f(y_i;\theta)$ as a function of $\theta$ given $y_1,\ldots,y_n$ and write
\begin{equation}
    L(\theta|y_1,\ldots,y_n) = \prod_{i=1}^{n}f(y_i;\theta)
\end{equation}
gives the \Term{likelihood function}, the likelihood that the population parameter is $\theta$ given the observed sample.

\subsection{ML principle}
The \Term{Maximum Likelihood (ML) principle} suggests that estimators of the unknwon parameters 
are obtained by \Emph{maximizing $L(\theta)$ with respect to $\theta$}. 
If the random variable $y$ is discrete, $\hat{\theta}$ yields the value of $\theta$ that maximizes
the probability of observing the sample that actually occurred.\\

Usually, $\hat{\theta}$ can be obtained by solving the likelihood equation
\begin{equation}
    \frac{\partial \log L(\theta)}{\partial \theta}|_{\hat{\theta}}=0
\end{equation}
where $\log L(\theta|y_1,\ldots,y_n) = \sum_{i=1}^{n}\log f(y_i;\theta)$.
Occationally the ML estimator is not unique and $\log L(\theta)$ may have only one global maximum,
but multiple local maxima.\\

The \Term{regularity conditions} are the following:
\begin{enumerate}
    \item The first three derivatives of $\log f(y|\theta)$ with respect to $\theta$ are continuous and finite
    for almost all $y$ and for all $\theta$.
    \item For all values of $\theta$, $|\frac{\partial^3\log f(y|\theta)}{\partial\theta_j\partial\theta_k\partial\theta_l}|$
    is limited by a function that has finite expectations.
    \item The domain of $y$ does not depend on $\theta$.
    \Info{For example of the uniform distribution, since the density function depends on $\theta_0$,
    the theorem does not hold.}
    \item $\theta$ is an interior point to the compact parameter space $\Theta$.
\end{enumerate}

\subsection{Bartlett Identities}
Define the \Term{score vector} $S(\theta)$ and the \Term{Hessian matrix} $H(\theta)$ as 
\begin{equation}
    S(\theta) = \frac{\partial\log L(\theta)}{\partial\theta} = \sum_{i=1}^{n}\frac{\partial\log f(y_i|\theta)}{\partial\theta}
    = \sum_{i=1}^{n}s_i(\theta)
\end{equation}
\begin{equation}
    H(\theta) = \frac{\partial^2\log L(\theta)}{\partial\theta\partial\theta'} 
    = \sum_{i=1}^{n}\frac{\partial^2\log f(y_i|\theta)}{\partial\theta\partial\theta'}
    = \sum_{i=1}^{n}H_i(\theta)
\end{equation}
\begin{itemize}
    \item The \Term{First Bartlett identity} states that $E[s_i(\theta_0)]=0$. Hence, $E[S(\theta_0)]=0$.
    \item The \Term{Second Bartlett identity} states that $Var[s_i(\theta_0)]=-E[H_i(\theta_0)]$.
    \begin{proof}
    \begin{equation*}
        Var\left[\frac{\partial\log f(y_i|\theta)}{\partial\theta}\right] = -E\left[\frac{\partial^2\log f(y_i|\theta_0)}{\partial\theta\partial\theta'}\right]
    \end{equation*}
    where $y_i$ is a random variable.
    \end{proof}
\end{itemize}
$Var[s_i(\theta)] = E[s_i(\theta_0)s_i(\theta_0)']$ defines the \Term{Fisher's information matrix}
denoted as $\mcI(\theta_0)$. Hence, the result $Var[s_i(\theta)] = E[s_i(\theta_0)s_i(\theta_0)']=-E[H_i(\theta_0)]$
is also called the \Term{information matrix identity}.
\begin{proof}
    \begin{align*}
        Var[s_i(\theta)] & = E[[s_i(\theta_0)-\underbrace{E[s_i(\theta_0)]}_{=0,\text{ First Bartlett}}]\times[s_i(\theta_0)-\underbrace{E[s_i(\theta_0)]}_{=0,\text{ First Bartlett}}]']\\
        & = E[s_i(\theta_0)s_i(\theta)']
    \end{align*}
\end{proof}

\section{Properties of MLE}
Under the assumed regularity conditions of MLE possesses the following properties:
\begin{enumerate}
    \item \Emph{Consistency} meaning $plim\hat{\theta} = \theta_0$
    \item \Emph{Asymptotic normality} meaning $\sqrt{n}(\hat{\theta}-\theta_0)\xrightarrow{d}\mcN(0,\mcI(\theta_0)^{-1})$.
    \Info{The goal is to estimate the $[\mcI(\theta_0)]^{-1}$ matrix (Fisher's information matrix)
    $\to$ need to go to a known distribution (normal).}
    \item \Emph{Asymptotic efficiency} meaning that if $\tilde{\theta}$ is a regular consistent asymptotically normal 
    estimator such that $\sqrt{n}(\tilde{\theta}-\theta_0)\xrightarrow{d}\mcN(0,\Omega)$,
    then $\Omega - [\mcI(\theta_0)]^{-1}$ is positive semi-definite, i.e.,
    under these RC, the MLE asymptotically achieves the \Term{Cramer-Rao} lower bound which is given by 
    $[\mcI(\theta_0)]^{-1}$.
    \Info{Information matrix is smaller than $\Omega$ in the matrix sense, and the information matrix is the lower bound.}
    \item \Emph{Invariance} meaning that if $c(\theta)$ is a continuous and continuously differentiable one-to-one function,
    the MLE of $\gamma = c(\theta_0)$ is $c(\hat{\theta})$.
\end{enumerate}
\begin{Example}
    In the Bernoulli case $\mcI(\theta_0)^{-1} = \theta_0(1-\theta_0)$,
    therefore, $\sqrt{n}(\hat{\theta}-\theta)\xrightarrow{d}\mcN(0,\theta_0(1-\theta_0))$.
    \begin{proof}
        We begin by writing the log-likelihood function,
        \begin{equation*}
            \log f(y_i|\theta) = y_i\log (\theta) + (1-y_i)\log(1-\theta)
        \end{equation*}
        Now finding the score vector,
        \begin{align*}
            s_i(\theta) & = \frac{\partial\log f(y_i|\theta)}{\partial\theta}\\
            & = \frac{y_i}{\theta} - \frac{(1-y_i)}{(1-\theta)}\\
            & = \frac{y_i}{\theta} - \frac{1}{1-\theta} + \frac{y_i}{1-\theta}\\
            & = y_i\left(\frac{1}{\theta}+\frac{1}{1-\theta}\right) - \frac{1}{\theta}\\
            & = y_i\left(\frac{1-\theta+\theta}{\theta(1-\theta)}\right) - \frac{1}{1-\theta}\\
            & = y_i\frac{1}{\theta(1-\theta)}-\frac{1}{1-\theta}\\
            & = \frac{y_i}{\theta(1-\theta)}-\frac{1}{1-\theta}
        \end{align*}
        Now finding the Fisher's information matrix,
        \begin{equation*}
            Var(s_i(\theta_0)) = Var\left[\frac{y_i}{\theta_0(1-\theta_0)}-\underbrace{\frac{1}{1-\theta_0}}_{=0,\text{ constant}}\right]
            = Var\left[\frac{y_i}{\underbrace{\theta_0(1-\theta_0)}_{\text{constant}}}\right]
            = \frac{1}{\theta_0^2(1-\theta_0)^2}\underbrace{Var[y_i]}_{(1)}
        \end{equation*}
        (1):
        \begin{equation*}
            \Var[y_i] = \underbrace{E[y_i^2]}_{(2)} - \underbrace{E[y_i]^2}_{(3)} 
            = \theta_0 - \theta_0^2 = \theta_0(1-\theta_0)
        \end{equation*}
        (2):
        \begin{equation*}
            E[y_i^2] = 1^2\underbrace{P(y_i=1|\theta_0)}_{=\theta_0} + 0^2\underbrace{p(y_i=0|\theta_0)}_{(1-\theta_0)}
            = \theta_0
        \end{equation*}
        (3):
        \begin{equation*}
            E[y_i] = 1\underbrace{P(y_i=1|\theta_0)}_{=\theta_0} + 0\underbrace{p(y_i=0|\theta_0)}_{(1-\theta_0)}
            = \theta_0
        \end{equation*}
        \begin{equation*}
            \Var(s_i(\theta_0)) = \frac{1}{\theta_0^2(1-\theta_0)^2}\theta_0(1-\theta_0)
            = \frac{1}{\theta_0(1-\theta_0)} = \mcI(\theta_0)
        \end{equation*}
        Now finding the inverse of the information matrix,
        \begin{equation*}
            \mcI(\theta_0)^{-1} = \theta_0(1-\theta_0)
        \end{equation*}
        We now conclude,
        \begin{equation*}
            \sqrt{n} (\hat{\theta}-\theta)\xrightarrow{d}\mcN(0, \theta_0(1-\theta_0))
        \end{equation*}
    \end{proof}
\end{Example}

Now, don't forget that $\theta_0$ is not known, so we want to estiamte the information matrix.

\subsection{Estimators of the information matrix}
There are three commonly used estimators of $\mcI(\theta_0)$:
\begin{itemize}
    \item \Emph{Expected Information}: If the form of the expected values of the second derivatives 
    of the log-likelihood function is known, then we can evaluate $\mcI(\theta)$ at $\hat{\theta}$. 
    \Info{This rarely happens in practice.}
    \item \Emph{Observed information}: Simply use $-\hat{H}(\hat{\theta})$.
    \begin{proof}
    \begin{align*}
        \mcI(\theta_0) & = Var[s_i(\theta_0)] = -E[H(\theta_0)]\\
        & = Var\left[\frac{\partial\log f(y_i|\theta_0)}{\partial\theta_0}\right]\\
        & = -E\left[\frac{\partial^2\log f(y_i|\theta_0)}{\partial\theta\partial\theta'}\right]\\
        & \underset{\text{avg.}}{=} \frac{1}{N}\sum_{i=1}^{n}\left(\frac{\partial^2\log f(y_i|\theta_0)}{\partial\theta\partial\theta'}\right)\\
        & = -\hat{H}(\hat{\theta})
    \end{align*}
    \end{proof}
    \item \Emph{Outer Product of the Gradient (OPG)}: Because of the information matrix identity,
    we can also use $n^{-1}\sum_{i=1}^{n}s_i(\hat{\theta})s_i(\hat{\theta})'$.
    \Info{The OPG is notorious for its poor performance with finite samples.}
    \begin{proof}
    \begin{align*}
        Var[s_i(\theta_0)] & = E[s_i(\theta_0)s_i(\theta)']\\
        & = \frac{1}{N}\sum_{i=1}^{N}s_i(\theta_0)s_i(\theta_0)'\\
        & = \frac{1}{N}\sum_{i=1}^{N}s_i(\hat{\theta})s_i(\hat{\theta})'
    \end{align*}
    \end{proof}
\end{itemize}

\section{Regressors}
Suppose the joint distribution of $y$ and $x$ depends on $\alpha_0$, giving
\begin{gather}
    f(y,x|\alpha_0) = f(y|x,\alpha_0)g(x|\alpha_0)\\
    f(y|x,\alpha_0) = \frac{f(y,x,\alpha_0)}{g(x,\alpha_0)}\\
    \underbrace{f(y,x,\alpha_0)}_{\text{joint}} = \underbrace{f(y|x,\alpha_0)}_{\text{conditional}}\underbrace{g(x,\alpha_0)}_{\text{marginal}}
\end{gather}
and suppose $\alpha_0$ can be divided into $\theta_0$ and $\delta_0$, so that with exogeneity of $x$, 
$f(y,x|\alpha_0) = f(y_i|x_i,\theta_0)g(x_i|\delta_0)$.
For an iid sample $\{(y_i,x_i)\}_{i=1}^{n}$,
\begin{equation}
    \log L(\theta,\delta|y_i,x_i) = \sum_{i=1}^{n}\log f(y_i,x_i|alpha) 
    = \sum_{i=1}^{n}\log f(y_i|x_i,\theta) + \sum_{i=1}^{n}\log g(x_i|\delta)
\end{equation}
\Info{
    We are under the assumption that we are only interested in $\theta_0$.
    Now this assumption that we don't care about $\delta$ makes our lives easier because 
    \begin{equation*}
        \frac{\partial L(\theta,\delta|y_i,x_i)}{\partial\theta}|_{\hat{\theta}}=0 \Leftrightarrow
        \frac{\partial \sum_{i=1}^{n}\log f(y_i|x_i,\theta_i)}{\partial\theta}|_{\hat{\theta}}=0
    \end{equation*}
    The part of $g(x,\delta)$ can be ignored now, only focus on the conditional density function.
}
However, now $E[-H_i(\theta_0)] = \mcI(\theta_0)$.

\section{Interpretation and inference in misspecified models}
If the likelihood function is \Emph{misspecified}, 
the MLE is generally \Emph{inconsistent} for the parameters of interest.
However, under the general conditions, $plim\hat{\theta} = \theta^*$, 
where the \Emph{pseudo-true value} $\theta^*$ minimizes the Kullback-Leibler divergence
where $f_0(y)$ is the true distribution of data.
\begin{definition}
    \textbf{Kullback-Leibler divergence}, also called relative entropy, 
    is a measure of how one probability distribution is different from a second,
    reference probability distribution.
\end{definition}
This means that the MLE leads to the best approximation in the Kullback-Leibner sense
to the true density. However, because Fisher's information matrix identity does not hold,
the asymptotic covariance matrix is given by:
\begin{equation}
    A^{-1}BA{-1}\quad A = E[-H_i(\theta^*)]\quad B = E[s_i(\theta^*)s_i(\theta^*)']
\end{equation}
\Info{$f(y|\theta^*)$ is the best approximation given the wrong space but is still wrong.}

\section{Hypothesis testing}
Considering a general set of restrictions to be tested
\begin{equation}
    H_0 : h(\theta_0) = 0
\end{equation}
where 
\begin{itemize}
    \item $\theta$: vector of parameters in the model and the true value is given by $\theta_0$.
    \item $h(\theta)$ is a $d\times 1$ vector of restrictions.
    \item $\hat{\theta}$ is the unrestricted MLE where its value maximizes $L(\theta)$.
    \item $\tilde{\theta}$ is the restricted MLE where its value maximizes $L(\theta)$ and satisfies $h(\theta) = 0$.
\end{itemize}

\subsection{The Wald Test}
Compares $h(\hat{\theta})$ with $0$ since $h(\tilde{\theta}) = 0$.
The test statistic is
\begin{equation}
    \mcW = n\times h(\hat{\theta})'\left[G(\hat{\theta})'[\hat{\mcI(\theta)}]^{-1}G(\hat{\theta})\right]^{-1}h(\hat{\theta})
\end{equation}
where $G(\theta) = \frac{\partial h(\theta)}{\partial\theta}$.\\

Under the null hypothesis,
\begin{equation}
    \mcW \xrightarrow{d} \mcX^2(d)
\end{equation}
\KeyIdea{Wald test is not invariant to how restrictions are formulated.}
Equivalent non-linear ($\beta_0/(1-\alpha_0)=1$) and linear ($\beta_0+\alpha_0-1 = 0$) restriction may lead to different values of $\mcW$.

\subsection{The Likelihood Ratio Test}
Compare $L(\hat{\theta})$ and $L(\tilde{\theta})$. 
If $h(\theta_0) = 0$ then $L(\hat{\theta})$ should be close to $L(\tilde{\theta})$.
The test is base on the likelihood ratio $\lambda = \frac{\mcL(\tilde{\theta})}{\mcL(\hat{\theta})}$ with the test statistic 
\begin{equation}
    \mcL\mcR = -2\log(\lambda) = 2\{\log\mcL(\hat{\theta}) - \log\mcL(\tilde{\theta})\}
\end{equation}
\Info{We may encounter convergence problem when maximizing.}\\

Under the null hypothesis,
\begin{equation}
    \mcL\mcR \xrightarrow{d} \mcX^2(d)
\end{equation}

\subsection{The Langrage Multiplier or Score Test}
Compare $S(\tilde{\theta})$ with $0$ since $S(\hat{\theta}) = 0$.
The test statistic is 
\begin{equation}
    \mcL\mcM = S(\tilde{\theta})'[\hat{\mcI(\theta)}]^{-1}S(\tilde{\theta})\frac{1}{n}
\end{equation}

Under the null hypothesis, 
\begin{equation}
    \mcL\mcM\xrightarrow{d}\mcX^2(d)
\end{equation}
\Info{The LM test works best in real life application. 
It is under weaker regulated conditions but it's still difficult to compute the $S(\theta)$ function.}
